cmake_minimum_required(VERSION 3.16)
project(mini_redis VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "Mini Redis uses Linux epoll/eventfd. Build on Linux or WSL.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_executable(mini_redis src/main.cpp src/resp.cpp src/store.cpp src/server.cpp)
target_compile_options(mini_redis PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(mini_redis PRIVATE Threads::Threads)
option(ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(ENABLE_SANITIZERS)
  target_compile_options(mini_redis PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(mini_redis PRIVATE -fsanitize=address,undefined)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:mini_redis>)
  set_tests_properties(integration PROPERTIES TIMEOUT 60)
endif()
